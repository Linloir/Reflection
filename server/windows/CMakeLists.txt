cmake_minimum_required(VERSION 3.21)
project(ReflectionWindows VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(reflection_core STATIC src/core.cpp src/config.cpp src/protocol.cpp)
target_include_directories(reflection_core PUBLIC include)
if(MSVC)
    target_compile_options(reflection_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(reflection_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(reflection_core_tests tests/core_tests.cpp)
    target_link_libraries(reflection_core_tests PRIVATE reflection_core)
    add_test(NAME reflection_core_tests COMMAND reflection_core_tests)
endif()

if(WIN32)
    target_compile_definitions(reflection_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0a00 WINVER=0x0a00)
    add_library(reflection_windows STATIC src/crypto.cpp src/pen_backend.cpp src/server.cpp src/runtime.cpp)
    target_link_libraries(reflection_windows PUBLIC reflection_core user32 ws2_32 bcrypt)
    add_executable(reflection-host src/main.cpp)
    target_link_libraries(reflection-host PRIVATE reflection_windows shell32)
    if(MSVC)
        target_compile_options(reflection-host PRIVATE /W4 /permissive-)
    endif()
    if(BUILD_TESTING)
        add_executable(reflection_crypto_tests tests/crypto_tests.cpp)
        target_link_libraries(reflection_crypto_tests PRIVATE reflection_windows)
        add_test(NAME reflection_crypto_tests COMMAND reflection_crypto_tests)
        add_executable(reflection_runtime_tests tests/runtime_tests.cpp)
        target_link_libraries(reflection_runtime_tests PRIVATE reflection_windows)
        add_test(NAME reflection_runtime_tests COMMAND reflection_runtime_tests)
    endif()
    install(TARGETS reflection-host RUNTIME DESTINATION bin)
endif()
